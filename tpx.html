/**
 * @title Transparency Token (TPX) & The Trinity Dual-Ledger Governance Architecture
 * 
 * TPX is the master utility and consensus verification token across:
 *   1. TaxFunded Token (TFT) — Taxpayer recovery, municipal audit, government waste tracking
 *   2. OSINT Coin (OSINT) — Multi-vector open source intelligence & civil rights litigation forensics
 *   3. The Trinity Consensus Engine:
 *      - Gate 1: Physical Reality & Proof-of-Custody (SHA-256 evidence anchoring)
 *      - Gate 2: Stochastic Probability & Multi-Source Cross-Corroboration
 *      - Gate 3: Statutory Authority, Baseline Frequency & Mean Reversion Filtering
 */

contract TransparencyTokenTPX {
    string public constant name = "Transparency Token";
    string public constant symbol = "TPX";
    uint8 public constant decimals = 18;
    uint256 public totalSupply;

    address public immutable platformAdmin;

    // Hard-Capped Supply (Bitcoin-style fixed supply model per architecture specification)
    uint256 public constant MAX_SUPPLY = 21_000_000 * 10**18;

    mapping(address => uint256) public balanceOf;
    mapping(address => mapping(address => uint256)) public allowance;

    // The Trinity Evidence Verification Record
    struct TrinityRecord {
        bytes32 evidenceHash; // SHA-256 hash of physical document / municipal record
        string entityTarget;  // e.g. "City of Huntington Beach", "Mercy House"
        uint256 batesNumber;
        uint256 trinityGateScore; // 0 to 100%
        string statutoryCitation; // e.g. "Cal. Gov. Code § 1090", "42 U.S.C. § 12132"
        uint256 timestamp;
        address verifier;
    }

    mapping(bytes32 => TrinityRecord) public trinityLedger;

    event Transfer(address indexed from, address indexed to, uint256 value);
    event Approval(address indexed owner, address indexed spender, uint256 value);
    event TrinityEvidenceAnchored(bytes32 indexed evidenceHash, string entityTarget, uint256 trinityScore, uint256 mintedReward);

    modifier onlyAdmin() {
        require(msg.sender == platformAdmin, "TPX: Caller is not platform admin");
        _;
    }

    constructor() {
        platformAdmin = msg.sender;
        // Initial genesis mint for platform deployment and liquidity
        uint256 initialGenesis = 1_000_000 * 10**decimals;
        totalSupply = initialGenesis;
        balanceOf[msg.sender] = initialGenesis;
        emit Transfer(address(0), msg.sender, initialGenesis);
    }

    function transfer(address to, uint256 value) public returns (bool) {
        require(balanceOf[msg.sender] >= value, "TPX: Insufficient balance");
        balanceOf[msg.sender] -= value;
        balanceOf[to] += value;
        emit Transfer(msg.sender, to, value);
        return true;
    }

    function approve(address spender, uint256 value) public returns (bool) {
        allowance[msg.sender][spender] = value;
        emit Approval(msg.sender, spender, value);
        return true;
    }

    function transferFrom(address from, address to, uint256 value) public returns (bool) {
        require(balanceOf[from] >= value, "TPX: Insufficient balance");
        require(allowance[from][msg.sender] >= value, "TPX: Allowance exceeded");
        balanceOf[from] -= value;
        allowance[from][msg.sender] -= value;
        balanceOf[to] += value;
        emit Transfer(from, to, value);
        return true;
    }

    /**
     * @notice Anchor verified primary evidence that has passed The Trinity Gates.
     * Mints TPX utility tokens directly to the forensic investigator.
     */
    function anchorTrinityEvidence(
        bytes32 evidenceHash,
        string memory entityTarget,
        uint256 batesNumber,
        uint256 trinityGateScore,
        string memory statutoryCitation,
        address investigatorRewardRecipient,
        uint256 rewardAmount
    ) public onlyAdmin {
        require(trinityGateScore >= 80, "TPX: Must clear Trinity Gate 2 threshold (>=80%)");
        require(totalSupply + rewardAmount <= MAX_SUPPLY, "TPX: Max supply capped");

        trinityLedger[evidenceHash] = TrinityRecord({
            evidenceHash: evidenceHash,
            entityTarget: entityTarget,
            batesNumber: batesNumber,
            trinityGateScore: trinityGateScore,
            statutoryCitation: statutoryCitation,
            timestamp: block.timestamp,
            verifier: msg.sender
        });

        totalSupply += rewardAmount;
        balanceOf[investigatorRewardRecipient] += rewardAmount;

        emit TrinityEvidenceAnchored(evidenceHash, entityTarget, trinityGateScore, rewardAmount);
        emit Transfer(address(0), investigatorRewardRecipient, rewardAmount);
    }
}
